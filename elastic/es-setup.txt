# Einmalige Einrichtung in Elasticsearch 8.19 – in Kibana unter "Dev Tools" ausführen
# (alternativ per curl mit einem Admin-Zugang)

# 1) Aufbewahrung: monatlich neuer Index, Löschung nach 90 Tagen (IP-Adressen und Benutzernamen
#    sind personenbezogen – Frist mit dem Datenschutzbeauftragten abstimmen)
PUT _ilm/policy/wachalarm-logs
{
  "policy": {
    "phases": {
      "hot": {
        "actions": {
          "rollover": { "max_age": "30d", "max_primary_shard_size": "10gb" }
        }
      },
      "delete": {
        "min_age": "90d",
        "actions": { "delete": {} }
      }
    }
  }
}

# 3) Index-Template für den Data Stream logs-wachalarm-*
#    Nutzt die mitgelieferten Bausteine für Logs und ECS, Priorität über dem Standard-Template "logs" (100)
PUT _index_template/logs-wachalarm
{
  "index_patterns": ["logs-wachalarm-*"],
  "data_stream": {},
  "priority": 200,
  "composed_of": ["logs@mappings", "logs@settings", "ecs@mappings", "wachalarm@custom"],
  "_meta": { "description": "Sicherheits- und Stabilitätsereignisse Wachalarm IP-Web" }
}

# Hinweis Einzelknoten: Ohne zweiten Knoten bleibt das Replikat unzugewiesen (Status "yellow").
# Dann in Schritt 2 unter "settings" zusätzlich "index.number_of_replicas": 0 eintragen.

# 4) API-Key für Filebeat – darf nur in logs-wachalarm-* schreiben
#    Aus der Antwort "id" und "api_key" als "id:api_key" in den Filebeat-Keystore übernehmen.
POST _security/api_key
{
  "name": "filebeat-wachalarm",
  "role_descriptors": {
    "wachalarm_writer": {
      "cluster": ["monitor", "read_ilm", "read_pipeline"],
      "indices": [
        {
          "names": ["logs-wachalarm-*"],
          "privileges": ["create_doc", "auto_configure", "view_index_metadata"]
        }
      ]
    }
  },
  "metadata": { "application": "wachalarm" }
}

# 5) Kontrolle nach dem Start von Filebeat
GET _data_stream/logs-wachalarm-default
GET logs-wachalarm-default/_search
{
  "size": 5,
  "sort": [{ "@timestamp": "desc" }]
}
